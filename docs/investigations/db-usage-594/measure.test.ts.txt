async function exec(sql: string): Promise<void> {
  await db.prepare(sql).run();
}

async function measure(name: string, sql: string, ...binds: unknown[]): Promise<void> {
  try {
    const r = await db.prepare(sql).bind(...binds).all();
    results.push({ name, reads: r.meta.rows_read ?? -1, writes: r.meta.rows_written ?? -1, rows: r.results.length });
  } catch (e) {
    results.push({ name: name + " [ERR " + String(e).slice(0, 60) + "]", reads: -1, writes: -1, rows: 0 });
  }
}

async function measureWrite(name: string, sql: string, ...binds: unknown[]): Promise<void> {
  // 書き込みの文は、後続の計測を汚さないよう、一時的に実行して meta を取る(データは変わるので最後の方で流す)
  try {
    const r = await db.prepare(sql).bind(...binds).run();
    results.push({ name, reads: r.meta.rows_read ?? -1, writes: r.meta.rows_written ?? -1, rows: r.meta.changes ?? 0 });
  } catch (e) {
    results.push({ name: name + " [ERR " + String(e).slice(0, 60) + "]", reads: -1, writes: -1, rows: 0 });
  }
}
